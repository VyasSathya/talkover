# faster-whisper 1.2.1 (ctranslate2 4.8.3, onnxruntime 1.31.0).

numpy>=2.0

# Kokoro text to speech (onnxruntime; model files from the kokoro-onnx GitHub release, see engines/native/fetch_models.py)
